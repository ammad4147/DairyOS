from dataclasses import dataclass, field
from datetime import datetime, timezone


@dataclass
class FarmOperationalState:
    """
    Living operational state of a dairy farm.

    Represents the current condition
    derived from operational activity.

    This model does not store historical events.

    Events describe what happened.

    Operational state describes
    what is true now.
    """


    farm_id: str


    operational_date: str


    created_at: datetime = field(
        default_factory=lambda:
            datetime.now(timezone.utc)
    )


    milk_status: dict = field(
        default_factory=dict
    )


    feeding_status: dict = field(
        default_factory=dict
    )


    health_alerts: list = field(
        default_factory=list
    )


    breeding_status: dict = field(
        default_factory=dict
    )


    workforce_status: dict = field(
        default_factory=dict
    )


    open_tasks: list = field(
        default_factory=list
    )


    exceptions: list = field(
        default_factory=list
    )



    def record_milk_activity(
        self,
        shift: str,
        litres: float,
    ):
        """
        Update milk operational state.
        """


        self.milk_status[shift] = {

            "status": "completed",

            "litres": litres,

        }



    def record_feed_activity(
        self,
        feed_type: str,
        quantity_kg: float,
    ):
        """
        Update feeding operational state.
        """


        self.feeding_status[feed_type] = {

            "status": "completed",

            "quantity_kg": quantity_kg,

        }



    def add_health_alert(
        self,
        animal_id: str,
        observation: str,
        severity: str,
    ):
        """
        Add health exception requiring attention.
        """


        self.health_alerts.append(

            {

                "animal_id": animal_id,

                "observation": observation,

                "severity": severity,

            }

        )



    def add_exception(
        self,
        exception: dict,
    ):
        """
        Add operational exception.
        """


        self.exceptions.append(
            exception
        )



    def summary(
        self,
    ):
        """
        Provide current operational snapshot.
        """


        return {

            "farm_id": self.farm_id,

            "operational_date":
                self.operational_date,

            "milk_status":
                self.milk_status,

            "feeding_status":
                self.feeding_status,

            "health_alert_count":
                len(self.health_alerts),

            "open_task_count":
                len(self.open_tasks),

            "exception_count":
                len(self.exceptions),

        }

class FarmOperationsRuntime:
    """
    Operational transaction boundary for DairyOS.
    """


    def __init__(
        self,
        milk_repository=None,
        feed_repository=None,
        health_repository=None,
        breeding_repository=None,
        event_gateway=None,
        event_repository=None,
        event_bus=None,
    ):

        self.milk_repository = (
            milk_repository
            if milk_repository is not None
            else MemoryMilkRepository()
        )

        self.feed_repository = (
            feed_repository
            if feed_repository is not None
            else MemoryFeedRepository()
        )

        self.health_repository = (
            health_repository
            if health_repository is not None
            else MemoryHealthRepository()
        )

        self.breeding_repository = (
            breeding_repository
            if breeding_repository is not None
            else MemoryBreedingRepository()
        )

        self.event_gateway = (
            event_gateway
            if event_gateway is not None
            else OperationsEventGateway()
        )

        self.event_repository = event_repository

        self.event_bus = (
            event_bus
            if event_bus is not None
            else FarmOperationEventBus()
        )

        self.events = []


    def _publish_event(
        self,
        event,
    ):

        timeline_event = (
            self.event_gateway.publish(event)
        )

        self.events.append(
            timeline_event
        )

        if self.event_repository is not None:
            self.event_repository.add(event)

        self.event_bus.publish(event)

        return timeline_event


    def record_milk(
        self,
        animal_group,
        shift,
        litres,
        operator,
    ):

        record = MilkRecord(
            animal_group=animal_group,
            shift=shift,
            litres=litres,
            operator=operator,
        )

        saved = self.milk_repository.save(record)

        self._publish_event(
            FarmOperationEvent(
                event_type="milk_recorded",
                animal_id=animal_group,
                operator=operator,
                payload={
                    "litres": litres,
                    "shift": shift,
                },
            )
        )

        return saved


    def record_feed(
        self,
        animal_group,
        feed_type,
        quantity_kg,
        cost,
        operator,
    ):

        record = FeedRecord(
            animal_group=animal_group,
            feed_type=feed_type,
            quantity_kg=quantity_kg,
            cost=cost,
            operator=operator,
        )

        saved = self.feed_repository.save(record)

        self._publish_event(
            FarmOperationEvent(
                event_type="feed_distributed",
                animal_id=None,
                operator=operator,
                payload={
                    "group_name": animal_group,
                    "feed_type": feed_type,
                    "quantity_kg": quantity_kg,
                    "cost": cost,
                },
            )
        )

        return saved


    def record_health(
        self,
        animal_id,
        observation,
        severity,
        reported_by,
    ):

        record = HealthObservation(
            animal_id=animal_id,
            observation=observation,
            severity=severity,
            reported_by=reported_by,
            observation_type="HEALTH_OBSERVATION",
            notes=observation,
            operator=reported_by,
        )


        saved = (
            self.health_repository
            .save(record)
        )


        self._publish_event(
            FarmOperationEvent(

                event_type="health_observation_recorded",

                animal_id=animal_id,

                operator=reported_by,

                payload={
                    "observation": observation,
                    "severity": severity,
                },

            )
        )


        return saved


    def record_breeding(
        self,
        animal_id,
        event_type,
        result,
        technician,
    ):

        record = BreedingRecord(
            animal_id=animal_id,
            event_type=event_type,
            result=result,
            technician=technician,
        )

        return self.breeding_repository.save(record)


    def get_events(
        self,
    ):

        return self.events

class FarmOperationsService:
    """
    Provides operational summaries
    from actual farm records.

    This service connects:
    farm activity -> operational intelligence -> management view.
    """


    def __init__(
        self,
        milk_repository=None,
        feed_repository=None,
        health_repository=None,
        breeding_repository=None,
    ):

        self.milk_repository = (
            milk_repository
            if milk_repository
            else MemoryMilkRepository()
        )

        self.feed_repository = (
            feed_repository
            if feed_repository
            else MemoryFeedRepository()
        )

        self.health_repository = (
            health_repository
            if health_repository
            else MemoryHealthRepository()
        )

        self.breeding_repository = (
            breeding_repository
            if breeding_repository
            else MemoryBreedingRepository()
        )


    def daily_milk_total(
        self,
    ):

        return sum(
            record.litres
            for record in self.milk_repository.get_all()
        )


    def daily_feed_quantity(
        self,
    ):

        return sum(
            record.quantity_kg
            for record in self.feed_repository.get_all()
        )


    def daily_feed_cost(
        self,
    ):

        return sum(
            record.cost
            for record in self.feed_repository.get_all()
        )


    def health_attention_count(
        self,
    ):

        return len(
            self.health_repository.get_all()
        )


    def breeding_pending_count(
        self,
    ):

        return len(
            self.breeding_repository.get_all()
        )


    def attention_items(
        self,
    ):
        """
        Returns operational items requiring attention.
        """

        items = []

        for observation in self.health_repository.get_all():

            items.append(
                {
                    "type": "health",
                    "animal_id": observation.animal_id,
                    "issue": observation.observation_type,
                    "notes": observation.notes,
                    "operator": observation.operator,
                }
            )

        return items

class DashboardFactory:
    """
    Application composition boundary
    for dashboard services.
    """



    @staticmethod
    def create(
        runtime: ApplicationRuntime | None = None,
    ):


        if runtime is None:

            runtime = ApplicationRuntime()



        domain_adapter = DomainDashboardAdapter(

            operations_runtime=(
                runtime.farm_operations_runtime
            ),

            animal_repository=(
                runtime.animal_repository
            ),

        )



        event_adapter = EventDashboardAdapter(

            event_repository=(
                runtime.operational_event_repository
            ),

            operations_runtime=(
                runtime.farm_operations_runtime
            ),

        )



        assembler = DashboardAssembler(

            adapter=domain_adapter,

            event_adapter=event_adapter,

        )


        return FarmDashboardService(

            assembler=assembler

        )

class DomainDashboardAdapter:


    def __init__(
        self,
        operations_runtime: FarmOperationsRuntime | None = None,
        animal_repository=None,
        operations_service=None,
        milk_repository=None,
        feed_repository=None,
        animal_bridge=None,
    ):

        self.operations_runtime = (
            operations_runtime
            if operations_runtime is not None
            else FarmOperationsRuntime(
                milk_repository=milk_repository,
                feed_repository=feed_repository,
            )
        )


        self.operations_service = operations_service

        self.animal_repository = animal_repository


        self.animal_bridge = (
            animal_bridge
            if animal_bridge is not None
            else AnimalOperationalBridge()
        )



    def get_milk_total(self):

        if self.operations_service:

            return (
                self.operations_service
                .daily_milk_total()
            )


        return sum(
            record.litres
            for record in (
                self.operations_runtime
                .milk_repository
                .get_all()
            )
        )



    def get_feed_consumption(self):

        if self.operations_service:

            return (
                self.operations_service
                .daily_feed_quantity()
            )


        return sum(
            record.quantity_kg
            for record in (
                self.operations_runtime
                .feed_repository
                .get_all()
            )
        )



    def get_total_animals(self):

        if self.animal_repository is None:

            return 0


        return len(
            self.animal_repository.get_all()
        )



    def get_milking_animals(self):

        if self.animal_repository is None:

            return 0


        count = 0


        for animal in self.animal_repository.get_all():

            if (
                getattr(
                    animal,
                    "is_milking",
                    False,
                )
                or
                getattr(
                    animal,
                    "is_currently_milking",
                    False,
                )
            ):

                count += 1


        return count



    def get_dry_animals(self):

        return max(
            self.get_total_animals()
            -
            self.get_milking_animals(),
            0,
        )



    def get_pending_tasks(self):

        if self.operations_service:

            return (
                self.operations_service
                .breeding_pending_count()
            )


        return 0



    def get_overdue_tasks(self):

        return 0



    def get_animal_attention_count(self):

        return len(
            [
                state
                for state in (
                    self.animal_bridge
                    .get_all_states()
                )
                if state.attention_required
            ]
        )



    def get_animal_attention_items(self):

        items = []


        for state in (
            self.animal_bridge
            .get_all_states()
        ):

            if state.attention_required:

                items.append(
                    {
                        "animal_id": state.animal_id,
                        "reasons": state.attention_reason,
                        "milk_deviation":
                            state.milk_deviation_percentage,
                    }
                )


        return items
