from dataclasses import dataclass, field
from datetime import datetime, timezone

from .operational_schedule_state import OperationalScheduleState


@dataclass
class FarmOperationalState:

    farm_id: str

    operational_date: str

    created_at: datetime = field(
        default_factory=lambda: datetime.now(timezone.utc)
    )

    milk_status: dict = field(
        default_factory=dict
    )

    feeding_status: dict = field(
        default_factory=dict
    )

    health_alerts: list = field(
        default_factory=list
    )

    breeding_status: dict = field(
        default_factory=dict
    )

    workforce_status: dict = field(
        default_factory=dict
    )

    inventory_status: dict = field(
        default_factory=dict
    )

    equipment_status: dict = field(
        default_factory=dict
    )

    operational_freshness: dict = field(
        default_factory=dict
    )

    milk_production_summary: dict = field(
        default_factory=lambda: {
            "total_litres_today": 0,
            "milking_events_count": 0,
            "last_milking_time": None,
            "last_operator": None,
            "last_animal_id": None,
        }
    )

    open_tasks: list = field(
        default_factory=list
    )

    completed_tasks: list = field(
        default_factory=list
    )

    heads_up_notifications: list = field(
        default_factory=list
    )

    exceptions: list = field(
        default_factory=list
    )

    schedule_state: OperationalScheduleState | None = None


    def __post_init__(self):

        if self.schedule_state is None:

            self.schedule_state = OperationalScheduleState(
                schedule_date=self.operational_date
            )


    def record_freshness(
        self,
        area,
        timestamp=None,
        source=None,
    ):

        self.operational_freshness[area] = {
            "last_updated": timestamp,
            "source": source,
        }


    def record_event(
        self,
        event,
        payload=None,
    ):

        if isinstance(event, str):

            event_type = event
            event_payload = payload or {}
            animal_id = event_payload.get(
                "animal_id"
            )

        else:

            event_type = getattr(
                event,
                "event_type",
                None,
            )

            event_payload = getattr(
                event,
                "payload",
                {},
            )

            animal_id = getattr(
                event,
                "animal_id",
                None,
            )

            animal_id = (
                event_payload.get("animal_id")
                or animal_id
            )


        if event_type in (
            "milk_recorded",
            "milk_activity_recorded",
        ):

            self.record_milk_activity(
                shift=(
                    event_payload.get("shift")
                    or event_payload.get("session")
                ),
                litres=event_payload.get(
                    "litres",
                    0,
                ),
                operator=event_payload.get(
                    "operator"
                ),
                animal_id=animal_id,
                timestamp=event_payload.get(
                    "timestamp"
                ),
            )


        elif event_type in (
            "feed_distributed",
            "feed_recorded",
        ):

            self.record_feed_activity(
                feed_type=event_payload.get(
                    "feed_type"
                ),
                quantity_kg=event_payload.get(
                    "quantity_kg",
                    0,
                ),
            )


        elif event_type in (
            "health_observation_recorded",
            "health_recorded",
        ):

            self.add_health_alert(
                animal_id=animal_id,
                observation=event_payload.get(
                    "observation",
                    "",
                ),
                severity=event_payload.get(
                    "severity",
                    "warning",
                ),
            )


        elif event_type in (
            "breeding_activity_recorded",
            "breeding_recorded",
        ):

            self.record_breeding_activity(
                animal_id=animal_id,
                details=event_payload,
            )


        elif event_type == "workforce_activity_recorded":

            self.record_workforce_activity(
                metric_type=event_payload.get(
                    "metric_type"
                ),
                value=event_payload.get(
                    "value"
                ),
            )


        elif event_type == "inventory_status_recorded":

            self.record_inventory_status(
                inventory_type=event_payload.get(
                    "inventory_type"
                ),
                item=event_payload.get(
                    "item"
                ),
                details=event_payload.get(
                    "details",
                    {},
                ),
            )


        elif event_type == "equipment_status_recorded":

            self.record_equipment_status(
                equipment_id=event_payload.get(
                    "equipment_id"
                ),
                details=event_payload.get(
                    "details",
                    {},
                ),
            )


        elif event_type == "task_created":

            self.record_open_task(
                event_payload
            )


        elif event_type == "task_completed":

            self.record_completed_task(
                event_payload
            )


        return self


    def record_milk_activity(
        self,
        shift,
        litres,
        animal_id=None,
        operator=None,
        timestamp=None,
    ):

        shift = shift or "UNKNOWN"

        current = self.milk_status.get(
            shift,
            {
                "status": "in_progress",
                "litres": 0,
                "animals_milked": 0,
                "operators": [],
                "last_entry_time": None,
            }
        )

        current["status"] = "completed"

        current["litres"] = (
            current.get("litres", 0)
            +
            litres
        )

        if animal_id is not None:

            current["animals_milked"] = (
                current.get("animals_milked", 0)
                +
                1
            )

        if operator is not None:

            operators = current.get(
                "operators",
                [],
            )

            if operator not in operators:
                operators.append(operator)

            current["operators"] = operators

        if timestamp is not None:

            current["last_entry_time"] = timestamp

        self.milk_status[shift] = current

        self.schedule_state.complete_milk_checkpoint(
            shift
        )


    def record_feed_activity(
        self,
        feed_type,
        quantity_kg,
    ):

        self.feeding_status[feed_type] = {
            "status": "completed",
            "quantity_kg": quantity_kg,
        }

        self.schedule_state.complete_feed_checkpoint(
            feed_type
        )

    def record_inventory_status(
        self,
        inventory_type,
        item,
        details,
    ):

        self.inventory_status[inventory_type] = {
            "item": item,
            "status": "AVAILABLE",
            **details,
        }


    def record_equipment_status(
        self,
        equipment_id,
        details,
    ):

        equipment = {
            **details,
        }

        if equipment.get("maintenance_required") is True:
            equipment.setdefault(
                "operational_status",
                "ATTENTION",
            )

        elif equipment.get("operational") is False:
            equipment.setdefault(
                "operational_status",
                "ATTENTION",
            )

        else:
            equipment.setdefault(
                "operational_status",
                "NORMAL",
            )

        self.equipment_status[equipment_id] = equipment


    def record_breeding_activity(
        self,
        animal_id,
        details,
    ):

        self.breeding_status[animal_id] = {
            **details,
        }


    def record_workforce_activity(
        self,
        metric_type,
        value,
    ):

        self.workforce_status[metric_type] = value


    def add_health_alert(
        self,
        animal_id,
        observation,
        severity,
    ):

        self.health_alerts.append(
            {
                "animal_id": animal_id,
                "observation": observation,
                "severity": severity,
            }
        )


    def record_open_task(
        self,
        task,
    ):

        task_record = self._normalize_task(task)

        task_record["status"] = "OPEN"

        self.open_tasks.append(
            task_record
        )


        if task_record.get("priority") == "HIGH":

            self.add_heads_up_notification(
                {
                    "type": "HIGH_PRIORITY_TASK",
                    "task_id": task_record.get(
                        "task_id"
                    ),
                }
            )


        due_date = task_record.get(
            "due_date"
        )

        if due_date:

            try:

                due = datetime.fromisoformat(
                    due_date.replace(
                        "Z",
                        "+00:00",
                    )
                )

                if due < datetime.now(timezone.utc):

                    self.add_heads_up_notification(
                        {
                            "type": "OVERDUE_TASK",
                            "task_id": task_record.get(
                                "task_id"
                            ),
                        }
                    )

            except Exception:

                pass



    def record_completed_task(
        self,
        task,
    ):

        task_record = self._normalize_task(task)

        task_record["status"] = "COMPLETED"

        task_record["completed_at"] = (
            datetime.now(timezone.utc)
        )

        self.completed_tasks.append(
            task_record
        )

        self.open_tasks = [
            item
            for item in self.open_tasks
            if item.get("task_id")
            != task_record.get("task_id")
        ]


    def _normalize_task(
        self,
        task,
    ):

        task = task or {}

        return {

            "task_id": task.get(
                "task_id"
            ),

            "task_type": task.get(
                "task_type",
                "GENERAL",
            ),

            "title": task.get(
                "title"
            ),

            "description": task.get(
                "description",
                "",
            ),

            "assigned_to": task.get(
                "assigned_to"
            ),

            "priority": task.get(
                "priority",
                "NORMAL",
            ),

            "created_at": task.get(
                "created_at",
                datetime.now(timezone.utc),
            ),

            "due_date": task.get(
                "due_date"
            ),

        }


    def add_heads_up_notification(
        self,
        notification,
    ):

        self.heads_up_notifications.append(
            notification
        )


    def add_exception(
        self,
        exception,
    ):

        self.exceptions.append(
            exception
        )


    def milk_total(self):

        return sum(
            item.get(
                "litres",
                0,
            )
            for item in self.milk_status.values()
        )


    def feed_total(self):

        return sum(
            item.get(
                "quantity_kg",
                0,
            )
            for item in self.feeding_status.values()
        )


    def health_alert_count(self):

        return len(
            self.health_alerts
        )


    def heads_up_count(self):

        return len(
            self.heads_up_notifications
        )


    def health_status(self):

        if any(
            alert.get("severity")
            in (
                "critical",
                "severe",
            )
            for alert in self.health_alerts
        ):

            return "critical"


        if self.health_alerts:

            return "warning"


        return "normal"


    def operational_status(self):

        if (
            self.exceptions
            or self.open_tasks
            or self.heads_up_notifications
        ):

            return "attention_required"


        if self.health_status() == "critical":

            return "attention_required"


        return "normal"


    def refresh_operational_awareness(self):

        return self.operational_status()


    def summary(self):

        return {

            "farm_id": self.farm_id,

            "operational_date": self.operational_date,

            "milk_status": self.milk_status,

            "feeding_status": self.feeding_status,

            "breeding_status": self.breeding_status,

            "workforce_status": self.workforce_status,

            "inventory_status": self.inventory_status,

            "equipment_status": self.equipment_status,

            "open_tasks": self.open_tasks,

            "completed_tasks": self.completed_tasks,

            "heads_up_notifications":
                self.heads_up_notifications,

            "milk_total":
                self.milk_total(),

            "feed_total":
                self.feed_total(),

            "operational_status":

        }
